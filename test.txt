
function splitStringAtDollar(input: string): string[] {
  // Check if the string contains the character $
  if (input.includes('$')) {
    // Split the string at the first occurrence of $
    return input.split('$');
  } else {
    // Return the original string as a single element array if $ is not found
    return [input];
  }
}

public class Mapper {

    private static final ObjectMapper objectMapper = new ObjectMapper();

    // Map ParentDBO to ParentDTO
    public static ParentDTO mapToDTO(ParentDBO parentDBO) {
        ParentDTO parentDTO = objectMapper.convertValue(parentDBO, ParentDTO.class);

        // Manually map child objects if necessary
        if (parentDBO.getChildren() != null) {
            List<ChildDTO> children = parentDBO.getChildren().stream()
                    .map(child -> objectMapper.convertValue(child, ChildDTO.class))
                    .collect(Collectors.toList());
            parentDTO.setChildren(children);
        }

        return parentDTO;
    }

    // Map list of ParentDBO to list of ParentDTO
    public static List<ParentDTO> mapToDTOList(List<ParentDBO> parentDBOs) {
        return parentDBOs.stream()
                .map(Mapper::mapToDTO)
                .collect(Collectors.toList());
    }
}

                    .withDetail("SMTP", "Server is unreachable")

            Process process = Runtime.getRuntime().exec(command);
            BufferedReader reader = new BufferedReader(new InputStreamReader(process.getInputStream()));
            String line;
            while ((line = reader.readLine()) != null) {
                output.append(line).append("\n");
            }
            process.waitFor();
        } catch (IOException | InterruptedException e) {
            return "Error executing command: " + e.getMessage();
        }
        return output.toString();
    }
}
