def lambda_handler(event, context):
 
    bedrock = boto3.client("bedrock-runtime", region_name="us-east-1")
    bedrock_agent = boto3.client("bedrock-agent-runtime", region_name="us-east-1")

 
    model_id = "anthropic.claude-3-5-sonnet-20240620-v1:0"
    knowledge_base_id = "FFBCCZXCK9"   

 
    query = event.get("prompt", "Summarize the key points in my project documentation.")

    
    retrieval = bedrock_agent.retrieve(
        knowledgeBaseId=knowledge_base_id,
        retrievalQuery={"text": query},
        retrievalConfiguration={
            "vectorSearchConfiguration": {"numberOfResults": 3}
        }
    )

    # Combine the retrieved text chunks
    if retrieval.get("retrievalResults"):
        context_snippets = "\n\n".join(
            r["content"]["text"] for r in retrieval["retrievalResults"]
        )
    else:
        context_snippets = "No relevant context was found in the knowledge base."

 
    prompt = f"""
You are a helpful assistant. Use the context below to answer the user’s question in a concise and natural way. 
Do not just repeat the context verbatim. If the answer is in the context, explain it naturally. 
If the answer is not found, say you don’t know.

Context:
{context_snippets}

Question:
{query}
"""

 
    body = {
        "anthropic_version": "bedrock-2023-05-31",
        "messages": [
            {"role": "user", "content": [{"type": "text", "text": prompt}]}
        ],
        "max_tokens": 512
    }

    response = bedrock.invoke_model(
        modelId=model_id,
        body=json.dumps(body)
    )

    result = json.loads(response["body"].read())
    output_text = result["content"][0]["text"]

 
    return {
        "statusCode": 200,
        "body": json.dumps({"response": output_text})
    }

import java.util.*;

public class NameParser {

    private static final Set<String> PREFIXES = Set.of(
        "mr", "mrs", "ms", "miss", "dr", "prof", "rev", "sir", "madam"
    );

    private static final Set<String> SUFFIXES = Set.of(
        "jr", "sr", "ii", "iii", "iv", "phd", "esq", "md", "jd"
    );

    private static final List<String> CORPORATE_KEYWORDS = List.of(
        "inc", "inc.", "llc", "l.l.c.", "ltd", "ltd.", "corp", "corporation",
        "company", "co.", "plc", "gmbh", "sa", "s.a.", "lp", "llp", "pc"
    );

    public static ParsedName parse(String input) {
        if (input == null || input.isBlank()) {
            return new ParsedName("", "", "", "", "");
        }

        String name = input.trim();
        String lower = name.toLowerCase();

        // Detect corporations — store full name in "lastName"
        for (String corp : CORPORATE_KEYWORDS) {
            if (lower.contains(corp)) {
                return new ParsedName("", "", "", name, "");
            }
        }

        // Split tokens
        String[] tokens = name.split("\\s+");
        List<String> tokenList = new ArrayList<>(Arrays.asList(tokens));

        String prefix = "";
        String first = "";
        String middle = "";
        String last = "";
        String suffix = "";

        // Detect prefix
        if (!tokenList.isEmpty() && PREFIXES.contains(cleanToken(tokenList.get(0)))) {
            prefix = tokenList.remove(0);
        }

        // Detect suffix
        if (!tokenList.isEmpty() && SUFFIXES.contains(cleanToken(tokenList.get(tokenList.size() - 1)))) {
            suffix = tokenList.remove(tokenList.size() - 1);
        }

        // Assign name parts
        if (tokenList.size() == 1) {
            first = tokenList.get(0);
        } else if (tokenList.size() == 2) {
            first = tokenList.get(0);
            last = tokenList.get(1);
        } else if (tokenList.size() >= 3) {
            first = tokenList.get(0);
            last = tokenList.get(tokenList.size() - 1);
            middle = String.join(" ", tokenList.subList(1, tokenList.size() - 1));
        }

        return new ParsedName(prefix, first, middle, last, suffix);
    }

    private static String cleanToken(String token) {
        return token.toLowerCase().replaceAll("[^a-z0-9]", "");
    }

    // --- Data class ---
    public static class ParsedName {
        public final String prefix;
        public final String firstName;
        public final String middleName;
        public final String lastName;
        public final String suffix;

        public ParsedName(String prefix, String first, String middle, String last, String suffix) {
            this.prefix = prefix;
            this.firstName = first;
            this.middleName = middle;
            this.lastName = last;
            this.suffix = suffix;
        }

        @Override
        public String toString() {
            return String.format(
                "prefix=%s, first=%s, middle=%s, last=%s, suffix=%s",
                prefix, firstName, middleName, lastName, suffix
            );
        }
    }
}

import java.util.*;
import java.util.regex.*;

public class NameParser {

    private static final Set<String> PREFIXES = Set.of("mr", "mrs", "ms", "miss", "dr", "prof", "rev", "hon");
    private static final Set<String> SUFFIXES = Set.of("jr", "sr", "phd", "md", "ii", "iii", "iv");
    private static final String[] ORG_KEYWORDS = {
        "inc", "inc.", "llc", "l.l.c", "corp", "corporation", "company", "co.", "co", "ltd", "ltd.",
        "limited", "group", "holdings", "partners", "plc", "gmbh", "s.a.", "s.p.a.", "ag", "nv",
        "bv", "associates", "foundation", "trust", "enterprise", "enterprises"
    };

    public static Map<String, String> parse(String fullName) {
        Map<String, String> parts = new LinkedHashMap<>();
        if (fullName == null || fullName.isBlank()) return parts;

        fullName = fullName.trim().replaceAll("[\"']", "");

        // Detect likely organization first
        if (isOrganization(fullName)) {
            parts.put("organization", fullName);
            return parts;
        }

        // Split into tokens
        String[] tokens = fullName.replaceAll("[,]", "").split("\\s+");
        List<String> words = new ArrayList<>();
        for (String t : tokens) if (!t.isBlank()) words.add(t.trim());

        if (words.isEmpty()) return parts;

        // Check prefix
        if (PREFIXES.contains(words.get(0).toLowerCase())) {
            parts.put("prefix", words.remove(0));
        }

        // Check suffix
        if (SUFFIXES.contains(words.get(words.size() - 1).toLowerCase())) {
            parts.put("suffix", words.remove(words.size() - 1));
        }

        // Split remaining words
        if (words.size() == 1) {
            parts.put("first", words.get(0));
        } else if (words.size() >= 2) {
            parts.put("first", words.get(0));
            parts.put("last", words.get(words.size() - 1));
            if (words.size() > 2)
                parts.put("middle", String.join(" ", words.subList(1, words.size() - 1)));
        }

        return parts;
    }

    private static boolean isOrganization(String name) {
        String lower = name.toLowerCase();

        // Contains org keyword
        for (String k : ORG_KEYWORDS) {
            if (lower.contains(k)) return true;
        }

        // Has @ means email address
        if (lower.contains("@")) return true;

        // All uppercase and longer than 2 characters
        if (name.equals(name.toUpperCase()) && name.length() > 2) return true;

        // Has digits (often product or company code)
        if (name.matches(".*\\d.*")) return true;

        // "of" or "&" inside name often indicates org
        if (lower.contains(" of ") || lower.contains(" & ")) return true;

        return false;
    }

    // Demo
    public static void main(String[] args) {
        List<String> tests = List.of(
            "Dr. John A. Smith Jr.",
            "O'Neil, Patrick",
            "Acme Holdings LLC",
            "BOARD OF EDUCATION",
            "support@example.com",
            "Smith & Wesson"
        );

        for (String t : tests) {
            System.out.println(t + " -> " + parse(t));
        }
    }
}

import com.azure.core.credential.TokenRequestContext;
import com.azure.identity.ClientSecretCredential;
import com.azure.identity.ClientSecretCredentialBuilder;
import com.microsoft.graph.requests.GraphServiceClient;
import okhttp3.Request;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;

@Configuration
public class GraphClientConfig {

    @Value("${azure.ad.client-id}")
    private String clientId;

    @Value("${azure.ad.tenant-id}")
    private String tenantId;

    @Value("${azure.ad.client-secret}")
    private String clientSecret;

    @Bean
    public GraphServiceClient<Request> graphServiceClient() {
        ClientSecretCredential credential = new ClientSecretCredentialBuilder()
                .clientId(clientId)
                .tenantId(tenantId)
                .clientSecret(clientSecret)
                .build();

        return GraphServiceClient.builder()
                .authenticationProvider(request -> {
                    String token = credential.getToken(
                            new TokenRequestContext()
                                    .addScopes("https://graph.microsoft.com/.default"))
                            .block()
                            .getToken();

                    request.addHeader("Authorization", "Bearer " + token);
                })
                .buildClient();
    }
}

import com.microsoft.graph.models.Message;
import com.microsoft.graph.requests.GraphServiceClient;
import com.microsoft.graph.requests.MessageCollectionPage;

import java.util.ArrayList;
import java.util.List;

public class GraphInboxFetcher {

    private final GraphServiceClient<?> graphClient;

    public GraphInboxFetcher(GraphServiceClient<?> graphClient) {
        this.graphClient = graphClient;
    }

    public List<Message> getAllUnflaggedMessages(String userEmail) {
        List<Message> allMessages = new ArrayList<>();

        MessageCollectionPage page = graphClient.users(userEmail)
                .mailFolders("inbox")
                .messages()
                .buildRequest()
                .filter("flag/flagStatus ne 'flagged'")
                .select("id,subject,from,flag,receivedDateTime")
                .top(50) // up to 999 allowed
                .get();

        while (page != null) {
            allMessages.addAll(page.getCurrentPage());

            if (page.getNextPage() != null) {
                page = page.getNextPage().buildRequest().get();
            } else {
                page = null;
            }
        }

        return allMessages;
    }
}

@Service

    private final GraphServiceClient<Request> graphClient;

    public MailService(GraphServiceClient<Request> graphClient) {
        this.graphClient = graphClient;
    }

    public List<String> searchInbox(String userEmail, String searchText) {
        return graphClient.users(userEmail)
                .messages()
                .buildRequest()
                .filter("contains(subject,'" + searchText + "')")
                .top(10)
                .get()
                .getCurrentPage()
                .stream()
                .map(Message::subject)
                .collect(Collectors.toList());
    }

    qry TEXT;
    col_exists BOOLEAN;
BEGIN

        SELECT tablename
        FROM pg_tables
        WHERE schemaname = 'public'
    LOOP
        -- check if column created_date exists in this table
        SELECT EXISTS (
            SELECT 1
            FROM information_schema.columns
            WHERE table_schema = 'public'
              AND table_name = r.tablename
              AND column_name = 'created_date'
        )
        INTO col_exists;

        IF NOT col_exists THEN
            RAISE NOTICE 'Table "%" does not contain column created_date', r.tablename;
        ELSE
            qry := format(
                'SELECT %L AS table_name
                 WHERE EXISTS (
                     SELECT 1 FROM %I WHERE created_date::date = CURRENT_DATE
                 )',
                r.tablename, r.tablename
            );

            EXECUTE qry;
        END IF;
    END LOOP;
END $$;

def upload_to_s3(file_path, bucket_name, object_key):
    s3 = boto3.client('s3')

    try:
        with open(file_path, "rb") as f:
            response = s3.put_object(
                Bucket=bucket_name,
                Key=object_key,

def decrypt_env_value(enc_str: str, key: str) -> str:
    if not enc_str.startswith("ENC(") or not enc_str.endswith(")"):

public class CustomRoleConverter implements Converter<Jwt, Collection<GrantedAuthority>> {

    @Override
    public Collection<GrantedAuthority> convert(Jwt jwt) {
        Object rolesObj = jwt.getClaim("roles");
        if (rolesObj instanceof Collection<?>) {
            return ((Collection<?>) rolesObj).stream()
                .map(Object::toString)
                .map(SimpleGrantedAuthority::new)
                .collect(Collectors.toList());
        }
        return Collections.emptyList();
    }

@Injectable()
export class AuthInterceptor implements HttpInterceptor {
  constructor(private authService: AuthService) {}

  intercept(req: HttpRequest<any>, next: HttpHandler): Observable<HttpEvent<any>> {
    const token = this.authService.getToken();

    if (token) {
      const cloned = req.clone({
        setHeaders: {
          Authorization: `Bearer ${token}`
        }
      });
      return next.handle(cloned);
    }

    return next.handle(req);
  }
}

function splitStringAtDollar(input: string): string[] {
  // Check if the string contains the character $
  if (input.includes('$')) {
    // Split the string at the first occurrence of $
    return input.split('$');
  } else {
    // Return the original string as a single element array if $ is not found
    return [input];
  }
}

public class Mapper {

    private static final ObjectMapper objectMapper = new ObjectMapper();

    // Map ParentDBO to ParentDTO
    public static ParentDTO mapToDTO(ParentDBO parentDBO) {
        ParentDTO parentDTO = objectMapper.convertValue(parentDBO, ParentDTO.class);

        // Manually map child objects if necessary
        if (parentDBO.getChildren() != null) {
            List<ChildDTO> children = parentDBO.getChildren().stream()
                    .map(child -> objectMapper.convertValue(child, ChildDTO.class))
                    .collect(Collectors.toList());
            parentDTO.setChildren(children);
        }

        return parentDTO;
    }

    // Map list of ParentDBO to list of ParentDTO
    public static List<ParentDTO> mapToDTOList(List<ParentDBO> parentDBOs) {
        return parentDBOs.stream()
                .map(Mapper::mapToDTO)
                .collect(Collectors.toList());
    }
}

  async exchangeCodeForTokens(code: string): Promise<void> {
    const body = new HttpParams()
      .set('grant_type', 'authorization_code')
      .set('code', code)
      .set('redirect_uri', window.location.origin)
      .set('client_id', authConfig.clientId);

    const headers = new HttpHeaders().set('Content-Type', 'application/x-www-form-urlencoded');

    try {
      const tokenResponse = await this.http.post<any>(authConfig.tokenEndpoint, body, { headers }).toPromise();
      this.oauthService.storeAccessTokenResponse(
        tokenResponse.access_token,
        tokenResponse.expires_in,
        tokenResponse.refresh_token,
        tokenResponse.scope,
        tokenResponse.id_token
      );
    } catch (error) {
      console.error('Error exchanging code for tokens', error);
      throw error;
    }
  }

    try {
      return await this.oauthService.loadUserProfile();
    } catch (error) {
      console.error('Error fetching user profile', error);
      throw error;
    }
  }

import org.springframework.boot.actuate.health.Health;

import javax.mail.MessagingException;
import javax.mail.Session;
import javax.mail.Transport;
import java.util.Properties;

@Component
public class SmtpHealthIndicator implements HealthIndicator {

    private final String smtpHost = "smtp.example.com"; // Replace with your SMTP server address
    private final int smtpPort = 587;                  // Common SMTP port (25, 465, or 587)
    private final String username = "your-email@example.com"; // Replace with your SMTP username
    private final String password = "your-password";         // Replace with your SMTP password

    @Override
    public Health health() {
        Properties properties = new Properties();
        properties.put("mail.smtp.auth", "true");
        properties.put("mail.smtp.starttls.enable", "true");
        properties.put("mail.smtp.host", smtpHost);
        properties.put("mail.smtp.port", smtpPort);

        Session session = Session.getInstance(properties);

        try (Transport transport = session.getTransport("smtp")) {
            // Attempt to connect to the SMTP server
            transport.connect(smtpHost, smtpPort, username, password);

            // If successful, return UP status
            return Health.up().withDetail("SMTP", "Server is reachable").build();
        } catch (MessagingException e) {
            // If an exception occurs, return DOWN status with the error message
            return Health.down()
                    .withDetail("SMTP", "Server is unreachable")
                    .withDetail("Error", e.getMessage())
                    .build();
        }
    }
}

import org.springframework.web.bind.annotation.*;
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;

@RestController
@RequestMapping("/api")
public class CurlExecutorController {

    @GetMapping("/execute")
    public String executeCurl(@RequestParam String url) {
        String command = "curl -s " + url;
        return executeCommand(command);
    }

    private String executeCommand(String command) {
        StringBuilder output = new StringBuilder();
        try {
            Process process = Runtime.getRuntime().exec(command);
            BufferedReader reader = new BufferedReader(new InputStreamReader(process.getInputStream()));
            String line;
            while ((line = reader.readLine()) != null) {
                output.append(line).append("\n");
            }
            process.waitFor();
        } catch (IOException | InterruptedException e) {
            return "Error executing command: " + e.getMessage();
        }
        return output.toString();
    }
}
